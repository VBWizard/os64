# The cascade (GARB.md, G2c), worked by hand from Cascade 4, Custom
# Properties 1, Media Queries 4 and Conditional Rules 3. Each PAGE is laid
# before a VIEWPORT; WANT is every element's author winners in document
# order, props in libgarb's order, ending (end). A page without a doctype is
# in quirks mode.

# Specificity outranks order; order breaks a tie.
VIEWPORT 800x600
PAGE
<!doctype html><style>

.a { color: green }
#x { color: blue }
p.a { color: yellow }
p { margin-top: 1px }
p { margin-top: 2px }
</style>
<p>one<p class=a>two<p id=x class=a>three
WANT
p: color: rgb(255, 0, 0); margin-top: 2px
p: color: rgb(255, 255, 0); margin-top: 2px
p#x: color: rgb(0, 0, 255); margin-top: 2px
(end)
END

# @media, @supports, and a style element's media. A grid and a flex
# container are laid out, wrapping or not, so both are SUPPORTED.
VIEWPORT 800x600
PAGE
<!doctype html><style>
p { color: red }
@media (max-width: 500px) { p { color: green } }
@media screen and (min-width: 600px) { p { color: blue } }
@supports (display: grid) { p { margin-top: 1px } }
@supports not (display: nonsense) { p { margin-left: 2px } }
@supports (foo: bar) { p { margin-right: 3px } }
@supports (display: table-cell) { p { margin-bottom: 4px } }
@supports not (display: inline-flex) { p { padding-left: 5px } }
@supports (display: contents) { p { padding-right: 6px } }
@supports (flex-wrap: wrap) { p { padding-top: 7px } }
@supports (flex-wrap: nowrap) { p { padding-bottom: 8px } }
</style>
<style media="print">p { width: 1px }</style>
<style media="(min-width: 700px)">p { height: 2px }</style>
<p>x
WANT
p: color: rgb(0, 0, 255); margin-top: 1px; margin-bottom: 4px; margin-left: 2px; padding-top: 7px; padding-right: 6px; padding-bottom: 8px; height: 2px
(end)
END

# Backgrounds in @supports: SUPPORTED is what is drawn. A gradient, a
# url() and the boxes it is placed in and cut to are; a background
# clipped to its text, a conic gradient and an image-set are read and not
# drawn (PILE3.md § Booked), so a page takes the fallback it wrote. A list
# of layers is drawn only when every layer is.
VIEWPORT 800x600
PAGE
<!doctype html><style>
@supports (background-image: linear-gradient(red, blue)) { p { margin-top: 1px } }
@supports (background-image: url(a.png)) { p { margin-right: 2px } }
@supports (background-clip: content-box) { p { margin-bottom: 3px } }
@supports (background-clip: text) { p { margin-left: 4px } }
@supports (background-image: conic-gradient(red, blue)) { p { padding-top: 5px } }
@supports (background: image-set("a.png" 1x)) { p { padding-right: 6px } }
@supports not (background-clip: text) { p { padding-bottom: 7px } }
@supports (background-clip: padding-box, text) { p { padding-left: 8px } }
@supports (background-image: url(a.png), conic-gradient(red, blue)) { p { width: 9px } }
@supports (background-image: url(a.png), linear-gradient(red, blue)) { p { height: 10px } }
</style>
<p>x
WANT
p: margin-top: 1px; margin-right: 2px; margin-bottom: 3px; padding-bottom: 7px; height: 10px
(end)
END

# @import (Cascade 4 § 2.1). A sheet the page's sheet imports is written
# after the page here, under a `@@SHEET <address>` line, and found by the
# address as written. An imported sheet's rules come BEFORE its importer's:
# a.css's blue loses to the page's own red, and a1.css (imported by a.css)
# comes before a.css, so a.css's margin-top wins over a1's. b.css's media
# does not hold at 800; c.css's supports() does; d.css's does not, and nor
# does f.css's, a flow-root laid out as a block rather than as written;
# e.css is imported after a rule, which makes the @import nothing.
VIEWPORT 800x600
PAGE
<!doctype html><style>
@charset "utf-8";
@import "a.css";
@import url(b.css) screen and (max-width: 500px);
@import "c.css" supports(display: block);
@import "d.css" supports(foo: bar);
@import "f.css" supports(display: flow-root);
p { color: red }
@import "e.css";
</style><p>x
@@SHEET a.css
@import "a1.css";
p { margin-top: 1px; color: blue }
@@SHEET a1.css
p { margin-left: 5px; margin-top: 9px }
@@SHEET b.css
p { margin-right: 2px }
@@SHEET c.css
p { padding-top: 3px }
@@SHEET d.css
p { padding-left: 4px }
@@SHEET e.css
p { width: 1px }
@@SHEET f.css
p { padding-bottom: 6px }
WANT
p: color: rgb(255, 0, 0); margin-top: 1px; margin-left: 5px; padding-top: 3px
(end)
END

# An imported sheet applies only where its importer does: the print sheet's
# import, and the import's own import, are nothing on a screen. A layer name
# on an @import is passed over (layers are cascaded as unlayered).
VIEWPORT 800x600
PAGE
<!doctype html><style media="print">@import "a.css";</style>
<style>@import "c.css" layer(base);</style><p>x
@@SHEET a.css
@import "a1.css";
p { margin-top: 1px }
@@SHEET a1.css
p { margin-left: 5px }
@@SHEET c.css
p { padding-top: 3px }
WANT
p: padding-top: 3px
(end)
END

# A custom property's empty value is a value, and its initial one is not:
# the "space toggle" (`--on: ;` substitutes nothing, `--off: initial` takes
# the fallback). And `unset` on a custom property inherits, as custom
# properties do (Cascade 4 § 7.3).
VIEWPORT 800x600
PAGE
<!doctype html><style>
div { --on: ; --off: initial; --u: blue }
p { --u: unset; color: var(--u, red); margin-top: var(--on) 3px; margin-bottom: var(--off, 4px) }
</style>
<div><p>x</div>
WANT
p: color: rgb(0, 0, 255); margin-top: 3px; margin-bottom: 4px
(end)
END
