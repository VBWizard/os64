# @font-face (Fonts 4 § 4), worked by hand: each PAGE laid before a
# VIEWPORT; WANT is each face garb_cascade_font_faces answers, in order —
# `"FAMILY" WEIGHTS STYLE [RANGES] sheet N`, then its sources, each
# `url(TEXT) [format(F)]` or `local(NAME)` — and `(N lost)` when any were,
# ending (end).

# The whole of a face: a string family; url() unquoted and quoted, with
# format() as a string and as a keyword; local() as a string and as words;
# a weight range written backwards; italic; ranges as a point, a range and
# a wildcard. Defaults: weight 400, normal, every code point.
VIEWPORT 800x600
PAGE
<!doctype html><style>
@font-face {
  font-family: "Pixel Operator";
  src: local("Pixel Operator"), local(Pixel   Op Mono), url(fonts/po.woff2) format("woff2"),
       url("fonts/po.ttf") format(truetype);
  font-weight: 700 300;
  font-style: italic;
  unicode-range: U+0025, U+0-7F, U+4??;
}
@font-face { font-family: Plain Words; src: url(a.ttf) }
</style>
WANT
"Pixel Operator" 300-700 italic U+25-25, U+0-7F, U+400-4FF sheet 0
  local(Pixel Operator)
  local(Pixel Op Mono)
  url(fonts/po.woff2) format(woff2)
  url(fonts/po.ttf) format(truetype)
"Plain Words" 400 normal sheet 0
  url(a.ttf)
(end)
END

# Only the @media that holds: a face inside one that does not is no face.
# A later rule of one family with the same weights, style and ranges
# replaces the earlier in its place; families compare without case; a
# different style is another face. The sheet a face came from.
VIEWPORT 800x600
PAGE
<!doctype html><style>
@font-face { font-family: Same; src: url(first) }
@font-face { font-family: Same; font-style: italic; src: url(italic) }
@media (max-width: 10px) { @font-face { font-family: Hidden; src: url(no) } }
@media (min-width: 10px) { @font-face { font-family: Shown; src: url(yes) } }
</style><style>
@font-face { font-family: SAME; src: url(second) }
</style>
