# @keyframes (Animations 1 § 3, Animations 2 § 3.2, Compat § 3.1), worked
# by hand: each PAGE laid before a VIEWPORT; WANT is, for every element
# whose winners name animations, in document order, each name and the
# keyframes garb_keyframes answers for it — `  OFFSET% [EASING]: prop:
# value; ...` — or `(no rule)`, ending (end).

# Offsets sorted; from and to are 0% and 100%; one block's two offsets are
# two keyframes; the implicit keyframes are not libgarb's.
VIEWPORT 800x600
PAGE
<!doctype html><style>
@keyframes fade { to { opacity: 1 } 50%, 75% { opacity: 0.5 } from { opacity: 0 } }
p { animation: fade 1s }
</style>
<p>x
WANT
p: fade
  0%: opacity: 0
  50%: opacity: 0.5
  75%: opacity: 0.5
  100%: opacity: 1
(end)
END

# The last rule of a name wins whole, whatever sheet or @media that holds
# it is in; one in an @media that does not hold is not kept. An identifier
# and a string are one name; case is not folded. A name no rule has.
VIEWPORT 800x600
PAGE
<!doctype html><style>
@keyframes a { from { color: red } to { color: blue } }
@keyframes "a" { to { opacity: 0 } }
@media (max-width: 10px) { @keyframes a { to { opacity: 1 } } }
@keyframes A { to { color: green } }
p { animation-name: a, A, missing }
</style>
<p>x
WANT
p: a
  100%: opacity: 0
p: A
  100%: color: rgb(0, 128, 0)
p: missing
  (no rule)
(end)
END

# Blocks at one offset with one easing are one keyframe, the last
# declaration of a property winning; a block that gives its own easing is
# another keyframe at that offset. `!important`, the animation properties
# and the custom properties are not kept; the timing function is the
# keyframe's easing, its first item, and the last one of a block counts.
VIEWPORT 800x600
PAGE
<!doctype html><style>
@keyframes k {
  50% { opacity: 0.2; color: red }
  50% { opacity: 0.4; margin-top: 1px !important }
  50% { animation-timing-function: steps(2, start), linear; animation-timing-function: ease-in; opacity: 0.9; animation-duration: 5s; --x: 1 }
  to { opacity: 1; animation: other 1s }
}
p { animation-name: k }
</style>
<p>x
WANT
p: k
  50%: color: rgb(255, 0, 0); opacity: 0.4
  50% ease-in: opacity: 0.9
  100%: opacity: 1
(end)
END

# Two blocks at one offset are one keyframe when their easings are one
# easing (Chrome, 10/14): a number past its sixth place tells two curves
# apart, though both print alike; a keyword curve is another easing than
# the same curve written out; steps are their count and where they jump
# however written.
VIEWPORT 800x600
PAGE
<!doctype html><style>
@keyframes e {
  50% { animation-timing-function: cubic-bezier(0, 0.0000001, 1, 1); opacity: 0.2 }
  50% { animation-timing-function: cubic-bezier(0, 0.0000002, 1, 1); opacity: 0.4 }
  50% { animation-timing-function: cubic-bezier(0, 0.0000001, 1, 1); color: red }
}
@keyframes s {
  20% { animation-timing-function: steps(2); opacity: 0.1 }
  20% { animation-timing-function: steps(2, jump-end); color: red }
  40% { animation-timing-function: step-start; opacity: 0.2 }
  40% { animation-timing-function: steps(1, jump-start); color: red }
  60% { animation-timing-function: step-end; opacity: 0.3 }
  60% { animation-timing-function: steps(1, end); color: red }
  80% { animation-timing-function: ease; opacity: 0.4 }
  80% { animation-timing-function: cubic-bezier(0.25, 0.1, 0.25, 1); color: red }
  80% { animation-timing-function: linear; opacity: 0.5 }
  80% { animation-timing-function: cubic-bezier(0, 0, 1, 1); opacity: 0.6 }
  90% { animation-timing-function: cubic-bezier(0, 0, 1, 1); opacity: 0.7 }
  90% { animation-timing-function: cubic-bezier(0, -0, 1, 1); color: red }
}
p { animation-name: e, s }
</style>
<p>x
WANT
p: e
  50% cubic-bezier(0, 0, 1, 1): color: rgb(255, 0, 0); opacity: 0.2
  50% cubic-bezier(0, 0, 1, 1): opacity: 0.4
p: s
  20% steps(2): color: rgb(255, 0, 0); opacity: 0.1
  40% step-start: color: rgb(255, 0, 0); opacity: 0.2
  60% step-end: color: rgb(255, 0, 0); opacity: 0.3
  80% ease: opacity: 0.4
  80% cubic-bezier(0.25, 0.1, 0.25, 1): color: rgb(255, 0, 0)
  80% linear: opacity: 0.5
  80% cubic-bezier(0, 0, 1, 1): opacity: 0.6
  90% cubic-bezier(0, 0, 1, 1): color: rgb(255, 0, 0); opacity: 0.7
(end)
END

# A selector that is not from, to or a percentage in [0%, 100%] drops its
# block; the rest stand.
VIEWPORT 800x600
PAGE
<!doctype html><style>
@keyframes s { 0% { opacity: 0 } 101% { opacity: 0.1 } 0 { opacity: 0.2 } 50%, x { opacity: 0.3 } 20%, { opacity: 0.4 } 100% { opacity: 1 } }
@keyframes none { to { opacity: 0 } }
@keyframes initial { to { opacity: 0 } }
@keyframes "none" { to { opacity: 0.5 } }
p { animation-name: s, "none" }
</style>
<p>x
WANT
p: s
  0%: opacity: 0
  100%: opacity: 1
p: none
  100%: opacity: 0.5
(end)
END

# var() in a keyframe is read against the element whose winners name the
# animation, with its own custom properties; one with none takes the
# fallback, and one whose var() has neither drops the declaration.

</style>

  100%: margin-top: 2px; margin-left: 10px
p: chip
  0%: margin-left: 3px
  100%: margin-left: 3px
(end)
